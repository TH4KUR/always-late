% ──────────────────────────────────────────────
\begin{frame}{The Research Question we started with}
\vfill
\centering
{\Large When a model's weights are rounded to fewer bits,\\[6pt]
do the internal features that \alert{explain} it\\[6pt]
still \alert{do the same job}?}
\vfill
\aside{27 languages $\cdot$ Gemma-2-2B (main) $\cdot$ Gemma-2-9B check $\cdot$ a fixed sparse autoencoder (SAE)}
\vfill
\end{frame}

% ──────────────────────────────────────────────
\begin{frame}{Background}
\begin{itemize}
\item \textbf{Quantization} stores weights with fewer bits (8 \ldots 4). Cheaper, but behaviour can change.
\item A \textbf{sparse autoencoder} splits a hidden state into a few active \textbf{features}.
\item Some features fire mostly for \textbf{one language}.
\item Earlier work asked only: does a feature still \emph{fire} the same way after quantization?
\end{itemize}
\vspace{8pt}
\takeaway{We also ask: does it still \alert{do} the same thing?}
\end{frame}

% ──────────────────────────────────────────────
\begin{frame}{A correction to earlier work}
\begin{columns}[c]
\column{0.5\textwidth}
\centering
\begin{tabular}{lcc}
\toprule
& \textbf{No BOS} & \textbf{With BOS}\\
\midrule
FP perplexity & 458.4 & \alert{16.1}\\
INT6 survival & 51\% & \alert{97.5\%}\\
INT7 survival & 81\% & \alert{$>$99.9\%}\\
\bottomrule
\end{tabular}
\column{0.46\textwidth}
We re-ran the earlier study's own code and matched its numbers.\\[8pt]
Its inputs left out the \textbf{start-of-text token} (BOS).\\[8pt]
That one token made the model look $28\times$ worse and half the features look ``dead''.
\end{columns}
\vspace{12pt}
\aside{All our results use the corrected inputs. We also retired our own first pipeline after finding flaws in it.}
\end{frame}

% ──────────────────────────────────────────────
\begin{frame}{Finding 1: language features matter, in every language}
\centering
\includegraphics[width=0.86\textwidth]{slide-harm}
\takeaway{\textbf{Left:} turning off a language's features hurts that language in \alert{all 27}.\quad
\textbf{Right:} the harm stays close to full precision down to 5 bits, then \alert{grows at 4 bits}.}
\end{frame}

% ──────────────────────────────────────────────
\begin{frame}{Finding 2: more suppression, more harm}
\begin{columns}[c]
\column{0.58\textwidth}
\includegraphics[width=\textwidth]{slide-dose}
\column{0.4\textwidth}
We turned features \textbf{down} step by step instead of fully off.\\[8pt]
Harm rises smoothly, in \alert{every language}.\\[8pt]
At 4 bits the same step hurts more.
\end{columns}
\vspace{6pt}
\aside{A graded response is evidence that these features have a real causal role.}
\end{frame}

% ──────────────────────────────────────────────
\begin{frame}{Finding 3: ``dead'' features still work}
\begin{columns}[c]
\column{0.6\textwidth}
\includegraphics[width=\textwidth]{slide-identity}
\column{0.38\textwidth}
\textbf{Orange:} the usual test says only \alert{13\%} of features survive at 4 bits.\\[8pt]
\textbf{Violet:} yet \alert{99.7\%} can still be recognised from what they do to the output.
\end{columns}
\vspace{8pt}
\takeaway{Whether a feature \emph{fires the same} and whether it \emph{works the same} are different questions.}
\end{frame}

% ──────────────────────────────────────────────
\begin{frame}{Finding 4: the change is not about the feature}
\begin{columns}[c]
\column{0.6\textwidth}
\includegraphics[width=\textwidth]{slide-swap}
\column{0.38\textwidth}
How similar is a feature's effect to its full-precision effect?\\[8pt]
A \textbf{random other direction} at the same places drifts \alert{almost equally}.
\end{columns}
\vspace{8pt}
\takeaway{The 4-bit drift comes from the model around the features, not from the features.}
\end{frame}

% ──────────────────────────────────────────────
\begin{frame}{Finding 5: fixing the features fixes little}
\begin{columns}[c]
\column{0.6\textwidth}
\includegraphics[width=\textwidth]{slide-repair}
\column{0.38\textwidth}
We put the full-precision values back into the 4-bit model.\\[8pt]
Restoring the language features recovers only \alert{2.6\%} of the lost quality.\\[8pt]
Restoring the whole layer recovers 39\%.
\end{columns}
\vspace{6pt}
\takeaway{The features are not where the compression damage sits.}
\end{frame}

% ──────────────────────────────────────────────
\begin{frame}{Finding 6: where does the change in an edit come from?}
\begin{columns}[c]
\column{0.46\textwidth}
\centering
\includegraphics[width=\textwidth]{slide-terms-design}
\column{0.52\textwidth}
\small
$D[\mathrm{early},\mathrm{late}]$ = how much the edit changes the output, with the early and late layers each at full precision (FP) or 4-bit (Q).
\begin{align*}
\textcolor{blue}{C} &= D[\mathrm{Q},\mathrm{FP}] - D[\mathrm{FP},\mathrm{FP}]\\
\textcolor{gold}{W} &= D[\mathrm{FP},\mathrm{Q}] - D[\mathrm{FP},\mathrm{FP}]\\
\textcolor{violet}{I} &= D[\mathrm{Q},\mathrm{Q}] - D[\mathrm{Q},\mathrm{FP}]\\
&\quad - D[\mathrm{FP},\mathrm{Q}] + D[\mathrm{FP},\mathrm{FP}]
\end{align*}
\aside{$C + W + I$ adds up exactly to the full change $D[\mathrm{Q},\mathrm{Q}] - D[\mathrm{FP},\mathrm{FP}]$.}
\end{columns}
\vspace{2pt}
\takeaway{We keep one edit exactly as at full precision and round \alert{only part} of the model.}
\end{frame}

% ──────────────────────────────────────────────
\begin{frame}{Finding 6: the answer depends on what we measure}
\centering
\includegraphics[width=0.8\textwidth]{slide-terms-bars}
\takeaway{One feature: \alert{processing (W)} matters most; a random edit changes as much.\\
A language's group: the \alert{new input (C)} matters most (19 of 27 at 4 bits).}
\end{frame}

% ──────────────────────────────────────────────
\begin{frame}{Which features stop firing? Weak ones.}
\begin{columns}[c]
\column{0.55\textwidth}
\includegraphics[width=\textwidth]{slide-tiers}
\column{0.43\textwidth}
Low-resource languages lose \alert{more} features.\\[6pt]
Language-selective features stop firing slightly more often.\\[6pt]
Both gaps come from \alert{weaker activations}: at equal strength they \alert{disappear}.
\end{columns}
\vspace{6pt}
\takeaway{Activation strength, not language or resource level, predicts which features stop firing.}
\end{frame}

% ──────────────────────────────────────────────
\begin{frame}{More features $\neq$ more resources}
\begin{columns}[c]
\column{0.6\textwidth}
\includegraphics[width=\textwidth]{slide-counts}
\column{0.38\textwidth}
\small
Lao, Yoruba and Khmer have more language-specific features than German or Spanish.\\[6pt]
A feature counts when it fires for \alert{one language} much more than the rest.\\[6pt]
Close relatives (Spanish--Portuguese, Russian--Ukrainian) \alert{share} features.
\end{columns}
\vspace{4pt}
\aside{The count shows how distinct a language looks to the model, not its resources (descriptive).}
\end{frame}

% ──────────────────────────────────────────────
\begin{frame}{Limitations}
\small
\begin{itemize}
\item \textbf{Reading comprehension (Belebele):} turning the features off does \alert{not} change accuracy.\\
\aside{Our held-out, confirmatory test. The features matter for next-word prediction, not clearly for this task.}
\item \textbf{One layer, one SAE:} earlier work finds language-specific units mostly in early and late layers; we tested only a middle layer (compute budget), a conservative place to look.
\item \textbf{Fixed-length inputs:} sentences are cut to a set length. The same cut text is used at every precision, so comparisons stay fair; testing complete sentences is future work.
\item \textbf{Mostly exploratory:} most FLORES analyses reuse sentences we had already examined.
\end{itemize}
\vspace{6pt}
\end{frame}

% ──────────────────────────────────────────────
\begin{frame}{Claims and Findings}
\begin{enumerate}
\item Language features help next-word prediction in \alert{all 27 languages}, on both Gemma sizes.
\item Turning them down step by step gives a \alert{smooth dose-response}.
\item The usual activation test \alert{understates} preservation: 13\% ``survive'', 99.7\% still work.
\item The 4-bit drift is \alert{generic}, and restoring the features repairs only \alert{2.6\%}.
\item Activation agreement, task performance and an edit's effect need \alert{separate measurements}.
\end{enumerate}
\end{frame}

\end{itemize}
