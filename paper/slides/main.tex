% ──────────────────────────────────────────────
\begin{frame}{The question}
\vfill
\centering
{\Large When a model's weights are rounded to fewer bits,\\[6pt]
do the internal features that \alert{explain} it\\[6pt]
still \alert{do the same job}?}
\vfill
\aside{27 languages $\cdot$ Gemma-2-2B (main) $\cdot$ Gemma-2-9B confirmation $\cdot$ Llama-3.2-1B cross-check $\cdot$ one fixed SAE per model}
\vfill
\end{frame}

% ──────────────────────────────────────────────
\begin{frame}{Background}
\begin{itemize}
\item \textbf{Quantization} rounds weights to fewer bits (8\ldots4). Reduces cost, but behaviour can change across languages (Marchisio et al., 2024).
\item A \textbf{sparse autoencoder (SAE)} decomposes a hidden state into a few active \textbf{features} from a learned dictionary (Gao et al., 2024).
\item Some features fire mostly for \textbf{one language}; editing them changes that language's output (Deng et al., 2025).
\item Duan (2026) measured whether feature \emph{activations} survive quantization.
\end{itemize}
\vspace{6pt}
\takeaway{We go further: does the feature still \alert{do the same thing} after quantization?}
\end{frame}

% ──────────────────────────────────────────────
\begin{frame}{Related work and our contribution}
\small
\begin{tabular}{p{3.4cm}p{5.2cm}p{4cm}}
\toprule
\textbf{Work} & \textbf{What they measure} & \textbf{Gap}\\
\midrule
Marchisio et al.~(2024) & Task outputs across languages after quantization & No internal mechanism\\
Deng et al.~(2025) & Language-selective SAE features; causal edits & No quantization\\
Duan (2026) & Activation survival ($R_1$) under quantization & No functional test\\
Makelov et al.~(2025) & Judge features by edits, not proxy metrics & Not applied to quantization\\
\bottomrule
\end{tabular}
\vspace{6pt}
\takeaway{We connect these: measure both \alert{activation survival} and \alert{functional preservation}, and show they come apart.}
\end{frame}

% ──────────────────────────────────────────────
\begin{frame}{Research questions}
\begin{description}
\item[\color{blue}RQ1] Which language features retain their \textbf{activation patterns} after quantization?
\item[\color{blue}RQ2] How does \textbf{editing} language features affect prediction and comprehension before and after quantization?
\item[\color{blue}RQ3] How well do \textbf{activation measures} predict whether a feature's \textbf{output effect} is preserved?
\item[\color{blue}RQ4] How much does the \textbf{same feature edit} change when the model is quantized, and what explains that change?
\end{description}
\end{frame}

% ──────────────────────────────────────────────
\begin{frame}{Experimental setup}
\begin{columns}[T]
\column{0.48\textwidth}
\textbf{Models \& SAEs}
\begin{itemize}\small
\item Gemma-2-2B, layer-12, 16k JumpReLU SAE
\item Gemma-2-9B, layer-20, 16k SAE
\item Llama-3.2-1B, layer-12, TopK SAE ($k{=}48$)
\end{itemize}
\vspace{4pt}
\textbf{Data}
\begin{itemize}\small
\item 27 FLORES-200 languages (9 per resource tier)
\item 300 aligned devtest sentences per language
\item Belebele: 385 questions $\times$ 18 languages
\end{itemize}

\column{0.48\textwidth}
\textbf{Quantizers}
\begin{itemize}\small
\item Legacy RTN (reproducing Duan)
\item Matched RTN (one scale per channel)
\item GPTQ (calibrated, 221k tokens)
\end{itemize}
\vspace{4pt}
\textbf{Feature selection}
\begin{itemize}\small
\item 8,086 language-selective features (2B)
\item Top $K{=}64$ per language (``K64 group'')
\item Activity-matched control groups
\end{itemize}
\end{columns}
\end{frame}

\begin{frame}{Methodology: the core test}
\centering
\begin{tikzpicture}[
  box/.style={draw=blue, line width=0.8pt, rounded corners=4pt, fill=panel, text=ink,
              text width=2.55cm, minimum height=1.55cm, align=center, font=\small},
  arr/.style={-{Stealth[length=6pt]}, line width=1pt, color=muted}]
\node[box] (a) {Pick the \textbf{64 features} that fire most for one language};
\node[box, right=0.45cm of a] (b) {\textbf{Zero-ablate} (set $a'_f = 0$)};
\node[box, right=0.45cm of b] (c) {Measure rise in \textbf{next-token loss} (NLL)};
\node[box, right=0.45cm of c] (d) {Subtract rise from \textbf{matched controls}};
\draw[arr] (a) -- (b); \draw[arr] (b) -- (c); \draw[arr] (c) -- (d);
\end{tikzpicture}

\vspace{8pt}
\[
\text{Excess harm: } E_\ell^b = \Delta\text{NLL}(\text{group}_\ell) - \overline{\Delta\text{NLL}(\text{controls}_\ell)}, \qquad \text{Effect ratio: } \rho = E^{(b)}/E^{(\text{FP})}
\]

\vspace{4pt}
\begin{tikzpicture}
\node[draw=gold, line width=0.8pt, rounded corners=4pt, fill=panel, text=ink, align=center, inner sep=7pt, font=\small]
{Repeat at FP and with weights rounded to \textbf{8, 7, 6, 5, 4 bits}};
\end{tikzpicture}
\end{frame}

\begin{columns}[c]
\column{0.5\textwidth}
\centering
\begin{tabular}{lcc}
\toprule
& \textbf{No BOS} & \textbf{With BOS}\\
\midrule
FP perplexity & 458.4 & \alert{16.1}\\
INT6 survival & 51\% & \alert{97.5\%}\\
INT7 survival & 81\% & \alert{$>$99.9\%}\\
\bottomrule
\end{tabular}
\column{0.46\textwidth}
We reproduced Duan's numbers with their code (INT6: 51.15\% vs.\ 51.30\%).\\[6pt]
The gap: their Gemma inputs omitted the \textbf{beginning-of-sequence (BOS) token}.\\[6pt]
One token shifted perplexity $28\times$ and halved apparent feature survival.
\end{columns}
\vspace{8pt}
\aside{All subsequent results use corrected inputs. We also retired our own first pipeline after finding separate flaws.}
\end{frame}

\begin{frame}{RQ1: Which activations survive?}
\begin{columns}[c]
\column{0.55\textwidth}
\includegraphics[width=\textwidth]{slide-tiers}
\column{0.43\textwidth}
\textbf{Activation survival:}
\[
R_1 = \operatorname{corr}_t\!\big(a_f^{\text{FP}}(t),\, a_f^{b}(t)\big)
\]
Rule: $R_1 > 0.9$.\\[4pt]
Low-resource languages lose \alert{more} features; selective features survive slightly less often.\\[4pt]
Both gaps trace to \alert{weaker activations} ($\rho_s = 0.35$ at INT6). Matching on strength \alert{removes} the gap.
\end{columns}
\vspace{2pt}
\aside{Selective features: median peak activation 9.8 vs.\ controls 14.2 at $S>5$.}
\end{frame}

% ──────────────────────────────────────────────
\begin{frame}{RQ1: Feature counts do not follow resource tier}
\begin{columns}[c]
\column{0.6\textwidth}
\includegraphics[width=\textwidth]{slide-counts}
\column{0.38\textwidth}
\small
Range: 73 (Portuguese) to 1,719 (English).\\[3pt]
Lao (464), Yoruba (453), Khmer (444) exceed most high-resource languages.\\[3pt]
Close relatives share: Portuguese--Spanish, Russian (83)--Ukrainian (84).\\[3pt]
$\rho_s = -0.07$ (count vs.\ tier).
\end{columns}
\vspace{2pt}
\aside{Count reflects how distinct a language looks to the model, not its resources (descriptive, post hoc).}
\end{frame}

\begin{frame}{RQ2: Dose--response confirms causal role}
\begin{columns}[c]
\column{0.58\textwidth}
\includegraphics[width=\textwidth]{slide-dose}
\column{0.4\textwidth}
Gradual suppression: $a'_f = \alpha\, a_f$.\\[6pt]
FP harm increases monotonically in \alert{all 27 languages}.\\[6pt]
Median harm: $0.022 \to 0.090 \to 0.169 \to 0.262$ nats/token.\\[6pt]
INT4 curve is \alert{higher at every step}.
\end{columns}
\vspace{4pt}
\aside{Dose and consistency rules hold; strict precision-wise monotonicity fails due to a small intermediate-width dip.}
\end{frame}

\begin{frame}{RQ3: Activation survival $\neq$ functional preservation}
\begin{columns}[c]
\column{0.6\textwidth}
\includegraphics[width=\textwidth]{slide-identity}
\column{0.38\textwidth}
\textbf{Effect fingerprint cosine:}
\[
W_1 = \frac{\mathbf{d}_f^{\text{FP}} \cdot \mathbf{d}_f^{b}}{\|\mathbf{d}_f^{\text{FP}}\|\;\|\mathbf{d}_f^{b}\|}
\]
At legacy INT4 ($n = 1{,}487$):\\[4pt]
\begin{tabular}{lr}
$R_1 > 0.9$ survival & \alert{13.2\%}\\
Identified by $W_1$ & \alert{99.7\%}\\
Median $W_1$ & $0.839$\\
\end{tabular}
\end{columns}
\vspace{4pt}
\takeaway{Activation cutoff declares 87\% ``dead''; almost all still produce recognizable output effects.}
\end{frame}

% ──────────────────────────────────────────────
\begin{frame}{RQ3: The 4-bit drift is generic (direction-swap control)}
\begin{columns}[c]
\column{0.6\textwidth}
\includegraphics[width=\textwidth]{slide-swap}
\column{0.38\textwidth}
Ablate a \textbf{different feature's direction} at the same positions and strengths.\\[6pt]
\begin{tabular}{lr}
Feature direction $W_1$ & 0.842\\
Swapped direction $W_1$ & 0.828\\
\textbf{Gap} & \alert{0.014}\\
\end{tabular}\\[6pt]
Gap is small and consistent at every precision.
\end{columns}
\vspace{4pt}
\takeaway{INT4 similarity loss is generic to downstream computation, not specific to the features.}
\end{frame}

% ──────────────────────────────────────────────
\begin{frame}{RQ3: Restoring features recovers almost nothing}
\begin{columns}[c]
\column{0.6\textwidth}
\includegraphics[width=\textwidth]{slide-repair}
\column{0.38\textwidth}
Patch FP activations into INT4:\\[4pt]
\small
\begin{tabular}{lr}
Language group K64 & \alert{2.6\%} $[2.4, 2.8]$\\
Matched control & 1.2\%\\
Full residual & 39.3\% $[38.5, 40.0]$\\
\end{tabular}\\[6pt]
\normalsize
GPTQ: 2.0\%.
\end{columns}
\vspace{4pt}
\takeaway{Language-group features carry $<$3\% of the INT4 compression damage.}
\end{frame}

\begin{frame}{RQ4: Decomposing the change in a fixed edit}
\begin{columns}[c]
\column{0.56\textwidth}
\includegraphics[width=\textwidth]{slide-terms-bars}
\column{0.42\textwidth}
Fix edit at FP direction and strength. Cross FP/quantized layers:
\begin{align*}
C &= D[\text{Q},\text{FP}] - D[\text{FP},\text{FP}]\\
W &= D[\text{FP},\text{Q}] - D[\text{FP},\text{FP}]\\
I &= D[\text{Q},\text{Q}] - C - W\\
  &\quad - D[\text{FP},\text{FP}]
\end{align*}
\vspace{-8pt}
\small
At INT4: median $|C| = 0.310$, $|W| = 0.104$ nats/token.\\[2pt]
\aside{For single-feature output vectors, $W$ is larger. Attribution depends on the edit and the outcome.}
\end{columns}
\end{frame}

\begin{frame}{Limitations}
\small
\begin{itemize}
\item \textbf{One layer, one SAE per model:} language-specific neurons concentrate in early/late layers (Tang et al., 2024; Kojima et al., 2024); middle-layer test is conservative.
\item \textbf{Fixed-length inputs:} truncation removes a length confound but may change internal representations. Paired comparisons keep both sides equal.
\item \textbf{Mostly exploratory:} most FLORES analyses reuse examined sentences. Only Belebele was confirmatory (null result).
\item \textbf{Llama inconsistency:} cross-family check failed (21/27), but model, SAE, tokenizer all changed.
\end{itemize}
\end{frame}

% ──────────────────────────────────────────────
\begin{frame}{Claims}
\begin{enumerate}
\item Language features contribute to next-token prediction in \alert{all 27 languages} on both Gemma sizes, with dose--response in every language.
\item The activation-survival cutoff \alert{understates} functional preservation: 13.2\% pass $R_1 > 0.9$, yet 99.7\% are identifiable by output effect $W_1$.
\item The INT4 effect-similarity loss is \alert{generic} (direction-swap gap 0.014), and restoring language-group activations recovers only \alert{2.6\%} $[2.4, 2.8]$ of the loss gap.
\item Activation agreement, task performance and an edit's output effect require \alert{separate measurements}: they answer different questions and give different answers.
\end{enumerate}
\end{frame}

% ──────────────────────────────────────────────
\begin{frame}{Reproducibility}
\begin{itemize}
\item Reproduced Duan's numbers with their released code before reporting the BOS issue
\item 9B predictions and code hashes frozen before any 9B data existed
\item Two defective 9B prediction criteria documented, not counted
\item Null results (Belebele, Llama, failed repair) reported with same detail as positives
\end{itemize}
